cmake_minimum_required(VERSION 3.16)
project(hri_vosk_providers)
find_package(ament_cmake REQUIRED)
find_package(hri_speech_providers REQUIRED)
find_package(capabilities2_runner REQUIRED)
find_package(hri_capability_interfaces REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tinyxml2 REQUIRED)
add_library(vosk_transcribe_speech_runner SHARED src/vosk_transcribe_speech_runner.cpp)
target_compile_features(vosk_transcribe_speech_runner PUBLIC cxx_std_17)
target_compile_options(vosk_transcribe_speech_runner PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(vosk_transcribe_speech_runner hri_speech_providers capabilities2_runner hri_capability_interfaces pluginlib rclcpp std_msgs)
target_link_libraries(vosk_transcribe_speech_runner tinyxml2::tinyxml2)
pluginlib_export_plugin_description_file(capabilities2_runner plugins.xml)
install(TARGETS vosk_transcribe_speech_runner LIBRARY DESTINATION lib)
install(DIRECTORY capability_providers DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/vosk_worker.py DESTINATION lib/${PROJECT_NAME} RENAME vosk_worker)
install(FILES scripts/vosk_core.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME vosk_core COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test -p test_*.py)
endif()
ament_package()

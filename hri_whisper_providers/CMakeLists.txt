cmake_minimum_required(VERSION 3.16)
project(hri_whisper_providers)
find_package(ament_cmake REQUIRED)
find_package(capabilities2_runner REQUIRED)
find_package(hri_capability_interfaces REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tinyxml2 REQUIRED)
add_library(whisper_transcribe_speech_runner SHARED src/whisper_transcribe_speech_runner.cpp)
target_compile_features(whisper_transcribe_speech_runner PUBLIC cxx_std_17)
target_compile_options(whisper_transcribe_speech_runner PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(whisper_transcribe_speech_runner capabilities2_runner hri_capability_interfaces pluginlib rclcpp std_msgs)
target_link_libraries(whisper_transcribe_speech_runner tinyxml2::tinyxml2)
pluginlib_export_plugin_description_file(capabilities2_runner plugins.xml)
install(TARGETS whisper_transcribe_speech_runner LIBRARY DESTINATION lib)
install(DIRECTORY capability_providers DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/whisper_worker.py DESTINATION lib/${PROJECT_NAME} RENAME whisper_worker)
install(FILES scripts/transcription_core.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME transcription_core COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/test -p test_*.py)
  add_executable(whisper_proxy_fixture test/whisper_proxy_fixture.cpp)
  target_compile_features(whisper_proxy_fixture PUBLIC cxx_std_17)
  ament_target_dependencies(whisper_proxy_fixture capabilities2_runner hri_capability_interfaces pluginlib rclcpp std_msgs)
  target_link_libraries(whisper_proxy_fixture tinyxml2::tinyxml2)
  install(TARGETS whisper_proxy_fixture DESTINATION lib/${PROJECT_NAME})
endif()
ament_package()
